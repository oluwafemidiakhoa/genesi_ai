```python
# ═══════════════════════════════════════════════════════════════════════
# DOWNLOAD MODELS FOR HUGGING FACE DEPLOYMENT
# ═══════════════════════════════════════════════════════════════════════

from google.colab import files
import os

print("="*70)
print("📥 DOWNLOADING MODELS TO YOUR COMPUTER")
print("="*70)

print("\nThese files will be downloaded to your Downloads folder:")
print("  1. best_model.pt - Genesis RNA transformer model")
print("  2. variant_classifier_rf.pkl - Random Forest classifier")
print("\nYou'll then upload these to your Hugging Face Space.")
print("\n" + "="*70)

# Determine which model was trained
if 'MODEL_PATH' in globals():
    model_path = MODEL_PATH
else:
    # Try to find the model
    quick_path = f"{DRIVE_DIR}/checkpoints/quick/best_model.pt"
    full_path = f"{DRIVE_DIR}/checkpoints/full/best_model.pt"

    if os.path.exists(quick_path):
        model_path = quick_path
    elif os.path.exists(full_path):
        model_path = full_path
    else:
        print("❌ ERROR: No trained model found!")
        print("Please run Cell 9 (Quick Training) or Cell 14 (Full Training) first.")
        raise FileNotFoundError("No model checkpoint found")

# Classifier path
classifier_path = f"{DRIVE_DIR}/results/variant_classifier_rf.pkl"

# Check both files exist
print("\n🔍 Checking files...")

if os.path.exists(model_path):
    model_size = os.path.getsize(model_path) / (1024 * 1024)
    print(f"✅ Genesis RNA model found: {model_path}")
    print(f"   Size: {model_size:.2f} MB")
else:
    print(f"❌ Model not found: {model_path}")

if os.path.exists(classifier_path):
    classifier_size = os.path.getsize(classifier_path) / (1024 * 1024)
    print(f"✅ Classifier found: {classifier_path}")
    print(f"   Size: {classifier_size:.2f} MB")
else:
    print(f"❌ Classifier not found: {classifier_path}")
    print("   Did Cell 24 complete successfully?")

# Download Genesis RNA model
if os.path.exists(model_path):
    print(f"\n📥 Downloading Genesis RNA model...")
    print(f"   This may take 1-2 minutes depending on file size...")
    files.download(model_path)
    print(f"✅ Downloaded: best_model.pt")

# Download classifier
if os.path.exists(classifier_path):
    print(f"\n📥 Downloading Random Forest classifier...")
    files.download(classifier_path)
    print(f"✅ Downloaded: variant_classifier_rf.pkl")

print("\n" + "="*70)
print("🎉 DOWNLOAD COMPLETE!")
print("="*70)

print("\n📋 Next Steps:")
print("\n1. Check your Downloads folder for:")
print("   • best_model.pt")
print("   • variant_classifier_rf.pkl")

print("\n2. Go to your Hugging Face Space:")
print("   https://huggingface.co/spaces/mgbam/genesis-rna-brca-classifier")

print("\n3. Upload both files:")
print("   • Click 'Add file' → 'Upload files'")
print("   • Select best_model.pt")
print("   • In 'Path' field, type: models/")
print("   • Click 'Commit changes'")
print("   • Repeat for variant_classifier_rf.pkl")

print("\n4. Wait for Space to rebuild (3-5 minutes)")

print("\n5. Test with variant: c.5266dupC (BRCA1)")
print("   Expected: Pathogenic with >95% confidence")

print("\n" + "="*70)
print("✨ Your trained models are ready for deployment!")
print("="*70)
```
