# 4. VERIFY DOWNLOADS
# ═══════════════════════════════════════════════════════════════════════

import os
import glob

print("📊 Downloaded Data:")
print("="*60)

# Check BRCA variants
brca_files = glob.glob('/content/genesi_ai/data/breast_cancer/brca_mutations/*.jsonl')
if brca_files:
    for f in brca_files:
        size = os.path.getsize(f) / (1024*1024)
        lines = sum(1 for _ in open(f))
        print(f"✅ {os.path.basename(f):<20} {lines:>5} variants ({size:.1f} MB)")
else:
    print("❌ BRCA variants not found")

# Check ncRNA
ncrna_files = glob.glob('/content/genesi_ai/data/human_ncrna/*.fa')
if ncrna_files:
    for f in ncrna_files:
        size = os.path.getsize(f) / (1024*1024)
        print(f"✅ {os.path.basename(f):<20} ({size:.1f} MB)")
else:
    print("❌ Human ncRNA not found")

print("="*60)
